    <script>
        let myChart;
        const chartData = {
            labels: Array(20).fill(''),
            datasets: [
                {
                    label: 'Heart Rate',
                    borderColor: '#ef4444',
                    borderWidth: 2,
                    data: Array(20).fill(70),
                    tension: 0.4,
                    pointRadius: 0
                },
                {
                    label: 'SpO2',
                    borderColor: '#3b82f6',
                    borderWidth: 2,
                    data: Array(20).fill(98),
                    tension: 0.4,
                    pointRadius: 0
                }
            ]
        };

        function showSection(id) {
            // Hide all sections
            document.querySelectorAll('.section').forEach(s => {
                s.classList.add('hidden');
                s.classList.remove('section-transition');
            });
            
            // Update nav styles
            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('bg-slate-800', 'text-white');
                btn.classList.add('text-slate-400');
            });
            
            // Show selected section with transition
            const activeSection = document.getElementById(id);
            activeSection.classList.remove('hidden');
            
            // Trigger animation restart
            void activeSection.offsetWidth; 
            activeSection.classList.add('section-transition');

            if(event) {
                event.currentTarget.classList.add('bg-slate-800', 'text-white');
                event.currentTarget.classList.remove('text-slate-400');
            }
            
            // Update Title
            const titles = { dashboard: 'Live Dashboard', history: 'Patient History', alerts: 'Emergency & Alerts', profile: 'Patient Profile' };
            document.getElementById('page-title').innerText = titles[id];
        }

        function initChart() {
            const ctx = document.getElementById('vitalsChart').getContext('2d');
            myChart = new Chart(ctx, {
                type: 'line',
                data: chartData,
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        y: { min: 40, max: 120, grid: { color: '#f1f5f9' } },
                        x: { display: false }
                    },
                    animation: false
                }
            });
        }

        function simulateData() {
            setInterval(() => {
                // Heartbeat simulation
                const hr = 70 + Math.floor(Math.random() * 15);
                const spo2 = 96 + Math.floor(Math.random() * 4);
                const temp = (98.2 + Math.random() * 0.8).toFixed(1);
                const glucose = 140 + Math.floor(Math.random() * 10);

                // Update UI numbers
                document.getElementById('val-bpm').innerText = hr;
                document.getElementById('val-spo2').innerText = spo2;
                document.getElementById('val-temp').innerText = temp;
                document.getElementById('val-glucose').innerText = glucose;

                // Update Chart
                chartData.datasets[0].data.shift();
                chartData.datasets[0].data.push(hr);
                chartData.datasets[1].data.shift();
                chartData.datasets[1].data.push(spo2);
                myChart.update();

                // Alert Check (Example: if HR > 100)
                if(hr > 100) {
                    document.getElementById('dashboard').classList.add('bg-red-50');
                } else {
                    document.getElementById('dashboard').classList.remove('bg-red-50');
                }
            }, 1500);
        }

        function simulateSOS() {
            const log = document.getElementById('sms-log');
            const items = document.getElementById('log-items');
            log.classList.remove('hidden');
            
            const time = new Date().toLocaleTimeString();
            const newItem = document.createElement('div');
            newItem.className = "flex justify-between items-center bg-slate-50 p-3 rounded-xl border-l-4 border-red-500 animate-pulse";
            newItem.innerHTML = `
                <div>
                    <p class="text-xs font-bold text-slate-400">${time}</p>
                    <p class="text-sm font-semibold text-slate-800">ALERT: Abnormal BPM (108) detected. Hospital contacted.</p>
                </div>
                <span class="text-[10px] bg-blue-100 text-blue-600 px-2 py-1 rounded font-bold">SENT VIA GSM</span>
            `;
            items.prepend(newItem);
        }

        window.onload = () => {
            initChart();
            simulateData();
        };
    </script>
